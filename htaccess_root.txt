# ───────────────────────────────────────────────
#  TWORIX 웹 루트 보안 설정
# ───────────────────────────────────────────────

# 디렉터리 목록 노출 금지 (폴더에 index가 없어도 파일 목록이 안 보이게)
Options -Indexes

# 1) 설정·비밀정보가 든 파일 직접 접근 차단
#    (브라우저로 telegram_config.php 등을 열어도 내용이 안 보이게)
<FilesMatch "^(telegram_config|mail_config)\.php$">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# 2) 밑줄(_)로 시작하는 내부 파일 차단
#    (_header.php, _footer.php, _auth_guard.php 등은 직접 열 필요 없음)
<FilesMatch "^_">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# 3) 데이터·백업·로그 확장자 직접 접근 차단
<FilesMatch "\.(json|log|bak|backup|old|tmp|sql|md|ini|sh)$">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# 4) 숨김 파일(.으로 시작) 차단 — .htaccess, .git 등
<FilesMatch "^\.">
  <IfModule mod_authz_core.c>
    Require all denied
  </IfModule>
  <IfModule !mod_authz_core.c>
    Order allow,deny
    Deny from all
  </IfModule>
</FilesMatch>

# 5) .git 폴더 통째로 차단 (GitHub 연동 시 소스 노출 방지)
RewriteEngine On
RewriteRule ^\.git - [F,L]
